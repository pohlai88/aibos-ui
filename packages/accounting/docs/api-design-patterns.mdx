# API Design Patterns & Best Practices

*A comprehensive guide to designing robust, scalable APIs for enterprise accounting systems*

## Introduction

This document outlines the API design patterns and best practices implemented in the AIBOS ERP system. These patterns ensure consistency, reliability, and maintainability across all API endpoints while providing excellent developer experience and robust error handling.

## Core Design Principles

Our APIs follow RESTful principles with clear resource-based URLs and HTTP methods.

### JWT-Based Authentication

Secure authentication using JSON Web Tokens with proper token management.

```typescript
interface AuthHeaders {
  'Authorization': string;        // Bearer <token>
  'x-tenant-id': string;         // Tenant identification
  'x-user-id': string;           // User identification
  'x-correlation-id'?: string;  // Request correlation

// Token validation middleware
@Injectable()
export class JwtAuthGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest();
    const token = this.extractTokenFromHeader(request);
    
    if (!token) {
      throw new UnauthorizedException('Token not provided');
    }
    
    try {
      const payload = this.jwtService.verify(token);
      request.user = payload;
      return true;
    } catch {
      throw new UnauthorizedException('Invalid token');
    }
  }
}
```

### Pagination Patterns

Consistent pagination across all list endpoints.

```typescript
interface PaginationQuery {
  page?: number;      // Default: 1
  limit?: number;    // Default: 20, Max: 100
  sort?: string;     // Field to sort by
  order?: 'asc' | 'desc'; // Sort order
  search?: string;   // Search term
  filters?: Record<string, unknown>; // Additional filters
}

// Pagination decorator
@Get('accounts')
async getAccounts(
  @Query() query: PaginationQuery,
  @Headers('x-tenant-id') tenantId: string
): Promise<PaginatedResponse<Account>> {
  const result = await this.accountService.getAccounts({
    tenantId,
    page: query.page || 1,
    limit: Math.min(query.limit || 20, 100),
    sort: query.sort,
    order: query.order || 'asc',
    search: query.search,
    filters: query.filters,
  });
  
  return {
    success: true,
    data: result.data,
    pagination: {
      page: result.page,
      limit: result.limit,
      total: result.total,
      totalPages: Math.ceil(result.total / result.limit),
      hasNext: result.page < Math.ceil(result.total / result.limit),
      hasPrevious: result.page > 1,
    },
    timestamp: new Date().toISOString(),
    requestId: this.generateRequestId(),
  };
}
```

## Advanced Patterns

### Idempotency

Support for idempotent operations to ensure safe retries.

```typescript
@Post('journal-entries')
async postJournalEntry(
  @Body() input: PostJournalEntryDto,
  @Headers('idempotency-key') idempotencyKey?: string,
  @Headers('x-tenant-id') tenantId: string
): Promise<JournalEntryResponseDto> {
  if (idempotencyKey) {
    const existing = await this.idempotencyService.get(idempotencyKey);
    if (existing) {
      return existing;
    }
  }

  const result = await this.accountingService.postJournalEntry({
    ...input,
    tenantId,
    idempotencyKey,
  });

  if (idempotencyKey) {
    await this.idempotencyService.store(idempotencyKey, result);
  }

  return result;
}
```

### Rate Limiting

Intelligent rate limiting based on user tiers and operation types.

```typescript
@Injectable()
export class RateLimitGuard implements CanActivate {
  constructor(
    private readonly redis: RedisService,
    private readonly userService: UserService
  ) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const userId = request.user.id;
    const endpoint = request.route.path;
    
    const userTier = await this.userService.getUserTier(userId);
    const limits = this.getRateLimits(userTier, endpoint);
    
    const current = await this.redis.get(`rate_limit:${userId}:${endpoint}`);
    if (current && parseInt(current) >= limits.requests) {
      throw new TooManyRequestsException('Rate limit exceeded');
    }
    
    await this.redis.incr(`rate_limit:${userId}:${endpoint}`);
    await this.redis.expire(`rate_limit:${userId}:${endpoint}`, limits.window);
    
    return true;
  }
}
```

### Request Correlation

Complete request tracing for debugging and monitoring.

```typescript
@Injectable()
export class CorrelationInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest();
    const correlationId = request.headers['x-correlation-id'] || this.generateCorrelationId();
    
    request.correlationId = correlationId;
    
    return next.handle().pipe(
      tap(() => {
        const response = context.switchToHttp().getResponse();
        response.setHeader('x-correlation-id', correlationId);
      }),
      catchError((error) => {
        this.logger.error(`Request failed: ${correlationId}`, error);
        throw error;
      })
    );
  }
}
```

## Error Handling Patterns

### Global Exception Filter

Centralized error handling with proper HTTP status codes.

```typescript
@Catch()
export class GlobalExceptionFilter implements ExceptionFilter {
  constructor(private readonly logger: Logger) {}

  catch(exception: unknown, host: ArgumentsHost): void {
    const ctx = host.switchToHttp();
    const response = ctx.getResponse<Response>();
    const request = ctx.getRequest<Request>();
    
    let status = HttpStatus.INTERNAL_SERVER_ERROR;
    let message = 'Internal server error';
    let code = 'INTERNAL_ERROR';
    
    if (exception instanceof HttpException) {
      status = exception.getStatus();
      message = exception.message;
      code = this.getErrorCode(exception);
    } else if (exception instanceof ZodError) {
      status = HttpStatus.BAD_REQUEST;
      message = 'Validation error';
      code = 'VALIDATION_ERROR';
    }
    
    const errorResponse: ApiError = {
      code,
      message,
      timestamp: new Date().toISOString(),
      requestId: request.correlationId,
      path: request.url,
    };
    
    this.logger.error(`API Error: ${request.correlationId}`, {
      error: exception,
      request: {
        method: request.method,
        url: request.url,
        headers: request.headers,
        body: request.body,
      },
    });
    
    response.status(status).json(errorResponse);
  }
}
```

### Business Rule Validation

Domain-specific validation with clear error messages.

```typescript
export class JournalEntryValidator {
  validateBusinessRules(command: PostJournalEntryCommand): void {
    // Validate minimum entries
    if (command.entries.length < 2) {
      throw new BusinessRuleViolationException(
        'JOURNAL_ENTRY_MIN_LINES',
        'Journal entry must have at least two lines',
        { minimumLines: 2, providedLines: command.entries.length }
      );
    }

    // Validate balance
    const totalDebits = command.entries.reduce((sum, e) => sum + e.debitAmount, 0);
    const totalCredits = command.entries.reduce((sum, e) => sum + e.creditAmount, 0);
    const difference = Math.abs(totalDebits - totalCredits);
    
    if (difference > 0.01) {
      throw new BusinessRuleViolationException(
        'JOURNAL_ENTRY_UNBALANCED',
        'Journal entry must be balanced (debits = credits)',
        { 
          totalDebits, 
          totalCredits, 
          difference,
          suggestion: 'Check amounts and ensure debits equal credits'
        }
      );
    }
  }
}
```

## Security Patterns

### Input Sanitization

Comprehensive input sanitization to prevent injection attacks.

```typescript
@Injectable()
export class SanitizationPipe implements PipeTransform {
  transform(value: any): any {
    if (typeof value === 'string') {
      return this.sanitizeString(value);
    }
    if (typeof value === 'object' && value !== null) {
      return this.sanitizeObject(value);
    }
    return value;
  }

  private sanitizeString(input: string): string {
    return input
      .replace(/[<>]/g, '') // Remove potential HTML tags
      .replace(/['"]/g, '') // Remove quotes
      .trim();
  }
}
```

### SQL Injection Prevention

Parameterized queries and input validation.

```typescript
@Injectable()
export class AccountRepository {
  async findByCode(accountCode: string, tenantId: string): Promise<Account | null> {
    // Use parameterized query to prevent SQL injection
    const query = `
      SELECT * FROM accounts 
      WHERE account_code = $1 AND tenant_id = $2
    `;
    
    const result = await this.dataSource.query(query, [accountCode, tenantId]);
    return result[0] || null;
  }
}
```
