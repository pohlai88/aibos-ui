# Database Schema Design Guide

*A comprehensive guide to the database architecture and schema design for the AIBOS ERP accounting system*

## Introduction

This document provides a detailed overview of the database schema design for the AIBOS ERP accounting system. The schema is designed to support enterprise-grade accounting operations with multi-tenancy, audit trails, and comprehensive financial reporting capabilities.

## Architecture Overview

### Database Technology Stack

- **Primary Database**: PostgreSQL 15+
- **Event Store**: PostgreSQL with JSONB for event data
- **Caching**: Redis for session management and query caching
- **Search**: PostgreSQL full-text search with GIN indexes
- **Backup**: Automated daily backups with point-in-time recovery

### Design Principles

1. **Multi-Tenancy**: Complete tenant isolation using Row-Level Security (RLS)
2. **Event Sourcing**: Immutable event store for complete audit trail
3. **ACID Compliance**: Full transactional integrity for financial data
4. **Performance**: Optimized indexes and query patterns
5. **Scalability**: Horizontal scaling support with sharding strategies

```sql
CREATE TABLE accounts (
```

### Query Optimization

Optimized query patterns for common operations.

```sql
-- Optimized trial balance query
CREATE OR REPLACE FUNCTION get_trial_balance(
  p_tenant_id UUID,
  p_period VARCHAR(7)
)
RETURNS TABLE (
  account_code VARCHAR(20),
  account_name VARCHAR(128),
  account_type VARCHAR(20),
  debit_balance DECIMAL(15,2),
  credit_balance DECIMAL(15,2),
  net_balance DECIMAL(15,2)
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    a.account_code,
    a.account_name,
    a.account_type,
    COALESCE(tb.debit_balance, 0) as debit_balance,
    COALESCE(tb.credit_balance, 0) as credit_balance,
    COALESCE(tb.net_balance, 0) as net_balance
  FROM accounts a
  LEFT JOIN trial_balance_cache tb ON (
    tb.tenant_id = p_tenant_id AND 
    tb.period = p_period AND 
    tb.account_code = a.account_code
  )
  WHERE a.tenant_id = p_tenant_id
    AND a.is_active = true
  ORDER BY a.account_code;
END;
$$ LANGUAGE plpgsql;

-- Optimized account balance query
CREATE OR REPLACE FUNCTION get_account_balance(
  p_tenant_id UUID,
  p_account_code VARCHAR(20),
  p_as_of_date DATE DEFAULT CURRENT_DATE
)
RETURNS DECIMAL(15,2) AS $$
DECLARE
  balance DECIMAL(15,2);
BEGIN
  SELECT COALESCE(SUM(
    CASE 
      WHEN debit_amount > 0 THEN debit_amount
      ELSE -credit_amount
    END
  ), 0)
  INTO balance
  FROM general_ledger_entries
  WHERE tenant_id = p_tenant_id
    AND account_code = p_account_code
    AND posting_date <= p_as_of_date;
  
  RETURN balance;
END;
$$ LANGUAGE plpgsql;
```

## Data Integrity and Constraints

### Business Rule Constraints

Database-level enforcement of business rules.

```sql
-- Account balance constraints based on account type
CREATE OR REPLACE FUNCTION validate_account_balance()
RETURNS TRIGGER AS $$
BEGIN
  -- Debit accounts (Assets, Expenses) should have positive balances
  IF NEW.account_type IN ('ASSET', 'EXPENSE') AND NEW.balance < 0 THEN
    RAISE EXCEPTION 'Debit account % cannot have negative balance', NEW.account_code;
  END IF;
  
  -- Credit accounts (Liabilities, Equity, Revenue) should have negative balances
  IF NEW.account_type IN ('LIABILITY', 'EQUITY', 'REVENUE') AND NEW.balance > 0 THEN
    RAISE EXCEPTION 'Credit account % cannot have positive balance', NEW.account_code;
  END IF;
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_validate_account_balance
  BEFORE INSERT OR UPDATE ON accounts
  FOR EACH ROW
  EXECUTE FUNCTION validate_account_balance();

-- Journal entry balance validation
CREATE OR REPLACE FUNCTION validate_journal_entry_balance()
RETURNS TRIGGER AS $$
DECLARE
  total_debits DECIMAL(15,2);
  total_credits DECIMAL(15,2);
BEGIN
  SELECT 
    COALESCE(SUM(debit_amount), 0),
    COALESCE(SUM(credit_amount), 0)
  INTO total_debits, total_credits
  FROM journal_entry_lines
  WHERE journal_entry_id = NEW.id;
  
  IF ABS(total_debits - total_credits) > 0.01 THEN
    RAISE EXCEPTION 'Journal entry % is not balanced: debits=%, credits=%', 
      NEW.journal_entry_id, total_debits, total_credits;
  END IF;
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_validate_journal_entry_balance
  AFTER INSERT OR UPDATE ON journal_entry_lines
  FOR EACH ROW
  EXECUTE FUNCTION validate_journal_entry_balance();
```

```sql
-- Archive old journal entries
CREATE OR REPLACE FUNCTION archive_old_journal_entries(
  p_tenant_id UUID,
  p_archive_date DATE DEFAULT CURRENT_DATE - INTERVAL '7 years'

RETURNS INTEGER AS $$
DECLARE
  archived_count INTEGER;
BEGIN
  -- Move old journal entries to archive
  WITH archived AS (
    DELETE FROM journal_entries
    WHERE tenant_id = p_tenant_id
      AND posted_at < p_archive_date
    RETURNING *
  )
  INSERT INTO journal_entries_archive
  SELECT * FROM archived;
  
  GET DIAGNOSTICS archived_count = ROW_COUNT;
  RETURN archived_count;
END;
$$ LANGUAGE plpgsql;
```

## Monitoring and Maintenance

### Performance Monitoring

Queries for monitoring database performance.

```sql
-- Query performance monitoring
CREATE OR REPLACE VIEW slow_queries AS
SELECT 
  query,
  calls,
  total_time,
  mean_time,
  rows,
  100.0 * shared_blks_hit / nullif(shared_blks_hit + shared_blks_read, 0) AS hit_percent
FROM pg_stat_statements
WHERE mean_time > 1000 -- Queries taking more than 1 second on average
ORDER BY mean_time DESC;

-- Index usage monitoring
CREATE OR REPLACE VIEW index_usage AS
SELECT 
  schemaname,
  tablename,
  indexname,
  idx_scan,
  idx_tup_read,
  idx_tup_fetch
FROM pg_stat_user_indexes
WHERE idx_scan = 0 -- Unused indexes
ORDER BY tablename, indexname;

-- Table size monitoring
CREATE OR REPLACE VIEW table_sizes AS
SELECT 
  schemaname,
  tablename,
  pg_size_pretty(pg_total_relation_size(schemaname||'.'||tablename)) AS size,
  pg_total_relation_size(schemaname||'.'||tablename) AS size_bytes
FROM pg_tables
WHERE schemaname = 'public'
ORDER BY pg_total_relation_size(schemaname||'.'||tablename) DESC;
```

### Maintenance Tasks

Automated maintenance procedures.

```sql
-- Vacuum and analyze maintenance
CREATE OR REPLACE FUNCTION maintenance_vacuum_analyze()
RETURNS VOID AS $$
BEGIN
  -- Vacuum all tables
  VACUUM ANALYZE accounts;
  VACUUM ANALYZE journal_entries;
  VACUUM ANALYZE journal_entry_lines;
  VACUUM ANALYZE event_store;
  VACUUM ANALYZE general_ledger_entries;
  
  -- Update table statistics
  ANALYZE accounts;
  ANALYZE journal_entries;
  ANALYZE journal_entry_lines;
  ANALYZE event_store;
  ANALYZE general_ledger_entries;
END;
$$ LANGUAGE plpgsql;

-- Index maintenance
CREATE OR REPLACE FUNCTION maintenance_reindex()
RETURNS VOID AS $$
BEGIN
  -- Reindex critical indexes
  REINDEX INDEX CONCURRENTLY idx_accounts_tenant_code_unique;
  REINDEX INDEX CONCURRENTLY idx_journal_entries_tenant_id_unique;
  REINDEX INDEX CONCURRENTLY idx_event_store_stream_version_unique;
END;
$$ LANGUAGE plpgsql;
```
