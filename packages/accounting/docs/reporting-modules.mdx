# Financial Reporting Modules

*A comprehensive guide to the reporting engine and financial intelligence capabilities of the AIBOS ERP system*

## Introduction

The AIBOS ERP reporting system provides comprehensive financial intelligence through automated report generation, real-time analytics, and sophisticated data visualization. Built for CFOs, accountants, and business stakeholders, it delivers the insights needed for informed decision-making.

### Data Collection

```typescript
class ReportDataCollector {
  async collectFinancialData(request: ReportRequest): Promise<FinancialData> {
    const [accounts, transactions, balances] = await Promise.all([
      this.getAccounts(request.tenantId),
      this.getTransactions(request.tenantId, request.period),
      this.getAccountBalances(request.tenantId, request.period)
    ]);

    return {
      accounts,
      transactions,
      balances,
      metadata: {
        collectionDate: new Date(),
        dataSource: 'primary',
        validationStatus: 'validated'
      }
    };
  }

  async validateData(data: FinancialData): Promise<ValidationResult> {
    const validation = await this.validationService.validate(data);
    if (!validation.isValid) {
      throw new DataValidationError(validation.errors);
    }
    return validation;
  }
}
```

### Report Calculation

```typescript
class ReportCalculator {
  calculateProfitAndLoss(data: FinancialData, period: string): ProfitAndLossStatement {
    const revenueAccounts = data.accounts.filter(a => a.accountType === 'REVENUE');
    const expenseAccounts = data.accounts.filter(a => a.accountType === 'EXPENSE');

    const revenue = this.calculateRevenue(revenueAccounts, data.transactions, period);
    const expenses = this.calculateExpenses(expenseAccounts, data.transactions, period);

    return {
      tenantId: data.tenantId,
      period,
      currencyCode: data.baseCurrency,
      generatedAt: new Date(),
      revenue: {
        grossRevenue: revenue.total,
        netRevenue: revenue.total - revenue.deductions,
        revenueAccounts: revenue.accounts
      },
      expenses: {
        totalExpenses: expenses.total,
        expenseAccounts: expenses.accounts
      },
      netIncome: revenue.total - expenses.total,
      grossProfit: revenue.total,
      operatingIncome: revenue.total - expenses.total
    };
  }

  calculateBalanceSheet(data: FinancialData, asOfDate: Date): BalanceSheet {
    const assets = this.categorizeAssets(data.accounts, data.balances);
    const liabilities = this.categorizeLiabilities(data.accounts, data.balances);
    const equity = this.categorizeEquity(data.accounts, data.balances);

    return {
      tenantId: data.tenantId,
      asOfDate,
      currencyCode: data.baseCurrency,
      generatedAt: new Date(),
      assets: {
        currentAssets: assets.current,
        fixedAssets: assets.fixed,
        totalAssets: assets.total
      },
      liabilities: {
        currentLiabilities: liabilities.current,
        longTermLiabilities: liabilities.longTerm,
        totalLiabilities: liabilities.total
      },
      equity: {
        equityAccounts: equity.accounts,
        totalEquity: equity.total
      },
      totalLiabilitiesAndEquity: liabilities.total + equity.total,
      isBalanced: Math.abs(assets.total - (liabilities.total + equity.total)) < 0.01
    };
  }
}
```

### Report Rendering

```typescript
class ReportRenderer {
  async renderReport(report: FinancialReport, format: ExportFormat): Promise<Buffer> {
    switch (format) {
      case 'PDF':
        return this.renderPDF(report);
      case 'EXCEL':
        return this.renderExcel(report);
      case 'CSV':
        return this.renderCSV(report);
      case 'JSON':
        return this.renderJSON(report);
      default:
        throw new UnsupportedFormatError(format);
    }
  }

  private async renderPDF(report: FinancialReport): Promise<Buffer> {
    const template = await this.getTemplate(report.type);
    const html = await this.renderHTML(report, template);
    return await this.htmlToPDF(html);
  }

  private async renderExcel(report: FinancialReport): Promise<Buffer> {
    const workbook = new ExcelJS.Workbook();
    const worksheet = workbook.addWorksheet(report.title);

    // Add headers
    worksheet.addRow(this.getHeaders(report));
    
    // Add data
    report.data.forEach(row => {
      worksheet.addRow(row);
    });

    // Apply formatting
    this.applyExcelFormatting(worksheet, report);

    return await workbook.xlsx.writeBuffer();
  }
}
```

```typescript
class ReportDeliveryService {
  async deliverReport(report: ReportResult, delivery: DeliveryMethod): Promise<void> {
    switch (delivery.type) {
      case 'EMAIL':
        await this.deliverByEmail(report, delivery.email);
        break;
      case 'FTP':
        await this.deliverByFTP(report, delivery.ftp);
        break;
      case 'API':
        await this.deliverByAPI(report, delivery.api);
        break;
      case 'STORAGE':
        await this.deliverToStorage(report, delivery.storage);
        break;
    }
  }

  private async deliverByEmail(report: ReportResult, email: EmailDelivery): Promise<void> {
    const emailContent = await this.generateEmailContent(report);
    await this.emailService.send({
      to: email.recipients,
      subject: email.subject,
      body: emailContent.body,
      attachments: [{
        filename: report.filename,
        content: report.data,
        contentType: report.contentType
      }]
    });
  }
}
```

## Performance Optimization

### Caching Strategy

```typescript
class ReportCacheService {
  async getCachedReport(key: string): Promise<ReportResult | null> {
    const cached = await this.cache.get(key);
    if (cached) {
      return JSON.parse(cached);
    }
    return null;
  }

  async cacheReport(key: string, report: ReportResult, ttl: number): Promise<void> {
    await this.cache.set(key, JSON.stringify(report), ttl);
  }

  async invalidateCache(pattern: string): Promise<void> {
    const keys = await this.cache.keys(pattern);
    await Promise.all(keys.map(key => this.cache.del(key)));
  }
}
```

### Query Optimization

```typescript
class OptimizedReportQueries {
  async getTrialBalanceOptimized(tenantId: string, period: string): Promise<TrialBalance> {
    // Use materialized view for better performance
    const query = `
      SELECT 
        account_code,
        account_name,
        account_type,
        debit_balance,
        credit_balance,
        net_balance
      FROM trial_balance_materialized_view
      WHERE tenant_id = $1 AND period = $2
      ORDER BY account_code
    `;
    
    const result = await this.database.query(query, [tenantId, period]);
    return this.mapToTrialBalance(result);
  }

  async getAccountBalancesOptimized(tenantId: string, asOfDate: Date): Promise<AccountBalance[]> {
    // Use indexed query with proper joins
    const query = `
      SELECT 
        a.account_code,
        a.account_name,
        a.account_type,
        COALESCE(SUM(gl.debit_amount), 0) as debit_balance,
        COALESCE(SUM(gl.credit_amount), 0) as credit_balance,
        COALESCE(SUM(gl.debit_amount - gl.credit_amount), 0) as net_balance
      FROM accounts a
      LEFT JOIN general_ledger_entries gl ON (
        a.account_code = gl.account_code AND 
        a.tenant_id = gl.tenant_id AND
        gl.posting_date <= $2
      )
      WHERE a.tenant_id = $1 AND a.is_active = true
      GROUP BY a.account_code, a.account_name, a.account_type
      ORDER BY a.account_code
    `;
    
    const result = await this.database.query(query, [tenantId, asOfDate]);
    return this.mapToAccountBalances(result);
  }
}
```

```typescript
class ExcelReportGenerator {
  async generateExcel(report: FinancialReport): Promise<Buffer> {
    const workbook = new ExcelJS.Workbook();
    
    // Add metadata
    workbook.creator = 'AIBOS ERP';
    workbook.created = new Date();
    workbook.modified = new Date();
    
    // Create worksheets
    const summarySheet = workbook.addWorksheet('Summary');
    const detailSheet = workbook.addWorksheet('Detail');
    
    // Add summary data
    this.addSummaryData(summarySheet, report);
    
    // Add detailed data
    this.addDetailData(detailSheet, report);
    
    // Apply formatting
    this.applyFormatting(workbook);
    
    // Add charts
    this.addCharts(workbook, report);
    
    return await workbook.xlsx.writeBuffer();
  }

  private addSummaryData(worksheet: Worksheet, report: FinancialReport): void {
    // Add headers
    worksheet.addRow(['Account Code', 'Account Name', 'Debit', 'Credit', 'Balance']);
    
    // Style headers
    const headerRow = worksheet.getRow(1);
    headerRow.font = { bold: true };
    headerRow.fill = {
      type: 'pattern',
      pattern: 'solid',
      fgColor: { argb: 'FFE0E0E0' }
    };
    
    // Add data
    report.data.forEach(row => {
      worksheet.addRow([
        row.accountCode,
        row.accountName,
        row.debitAmount,
        row.creditAmount,
        row.balance
      ]);
    });
    
    // Add totals
    const totalRow = worksheet.addRow([
      'TOTAL',
      '',
      report.totalDebits,
      report.totalCredits,
      report.totalDebits - report.totalCredits
    ]);
    
    totalRow.font = { bold: true };
    totalRow.fill = {
      type: 'pattern',
      pattern: 'solid',
      fgColor: { argb: 'FFF0F0F0' }
    };
  }
}
```

### Data Masking

```typescript
class DataMaskingService {
  maskSensitiveData(data: any, userRole: string): any {
    const maskingRules = this.getMaskingRules(userRole);
    
    return this.applyMaskingRules(data, maskingRules);
  }

  private getMaskingRules(userRole: string): MaskingRule[] {
    switch (userRole) {
      case 'VIEWER':
        return [
          { field: 'salary', mask: '***' },
          { field: 'ssn', mask: '***-**-****' },
          { field: 'bankAccount', mask: '****-****-****-****' }
        ];
      case 'ANALYST':
        return [
          { field: 'ssn', mask: '***-**-****' }
        ];
      case 'ADMIN':
        return []; // No masking for admin
      default:
        return this.getDefaultMaskingRules();
    }
  }
}
```
